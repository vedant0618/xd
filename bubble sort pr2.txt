Salary = []
size = int(input("Enter the array size: "))

for i in range(size):
    ele = float(input("Enter the element: "))
    Salary.append(ele)


def bubble_sort(arr):
    # Work on a copy so the original list stays unsorted for other menu choices
    data = arr.copy()
    n = len(data)
    for i in range(n - 1):
        for j in range(n - i - 1):
            if data[j] > data[j + 1]:
                data[j], data[j + 1] = data[j + 1], data[j]
    return data


def selection_sort(arr):
    # Use len(data) instead of relying on the global 'size' variable
    data = arr.copy()
    n = len(data)
    for i in range(n - 1):
        for j in range(i + 1, n):
            if data[i] > data[j]:
                data[i], data[j] = data[j], data[i]
    return data


def menu():
    print("\n1. Bubble sort")
    print("2. Selection sort")
    print("3. Exit")


while True:
    menu()
    choice = input("Enter your choice (1-3): ")

    if choice == '1':
        sorted_bubblesort = bubble_sort(Salary)
        print("Sorted bubble sort is:", sorted_bubblesort)
        print("The length of bubble sort is:", len(sorted_bubblesort))
        print("Top 5 salaries are:", sorted_bubblesort[-1:-6:-1])

    elif choice == '2':
        sorted_selectionsort = selection_sort(Salary)
        print("Sorted selection sort is:", sorted_selectionsort)
        print("The length of selection sort is:", len(sorted_selectionsort))
        print("Top 5 salaries are:", sorted_selectionsort[-1:-6:-1])

    elif choice == '3':
        print("Exit")
        break

    else:
        print("Invalid Choice")
